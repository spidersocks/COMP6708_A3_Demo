\begin{frame}{The problem}{\thesection}
  \begin{columns}[T]
    \begin{column}{0.62\textwidth}
      \textbf{Imagine you run a tutoring centre.}
      \vspace{0.35em}
      \begin{itemize}
        \item Parents message after hours: typhoons, fees, seats, trials, booking
        \item Staff cannot live on WhatsApp 24/7
        \item A wrong fee or hours answer costs trust --- fast
      \end{itemize}
      \vspace{0.45em}
      \begin{alertblock}{So you hired Group 15}
        A couple of weeks later we built \ctextbf{BrightNest}: grounded answers,
        structured actions, honest hand-offs.
      \end{alertblock}
    \end{column}
    \begin{column}{0.34\textwidth}
      \centering
      \bnlogo\\[0.35em]
      {\scriptsize BrightNest}\\[0.7em]
      \begin{tikzpicture}[font=\scriptsize, every node/.style={align=center}]
        \node[draw=maincolor, rounded corners=3pt, thick,

        };
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Technique: training-free RAG + tools}{\thesection}
  \textbf{Why training-free?} Centre docs change weekly $\Rightarrow$ fine-tuning is the wrong loop.
  \vspace{0.35em}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{RAG (self-built)}
      \begin{enumerate}\setlength\itemsep{0.15em}
        \item Chunk markdown KB (\texttt{content/})
        \item Embed with DashScope
        \item Store / query in Chroma
        \item Lexical rerank $\rightarrow$ top-$k$
        \item Generate with Qwen + citations
      \end{enumerate}
      \vspace{0.25em}
      {\scriptsize Retrieval score $s(q,c)=\mathrm{sim}(E(q),E(c))$ · inspector shows rank, $s$, full chunk.}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Hybrid router}
      \begin{itemize}\setlength\itemsep{0.15em}
        \item Hard rules: typhoon, fees, seats
        \item Soft prototypes for paraphrase
        \item Cheap LLM review \emph{only if} low confidence
        \item Else: \texttt{high\_confidence\_skip}
      \end{itemize}
      \vspace{0.35em}
      \pill{tools for truth}\quad\pill{RAG for narrative}\\[0.25em]
      \pill{escalate when human}
    \end{column}
  \end{columns}
\end{frame}
