\begin{frame}{Problem \& application scenario}{\thesection \, \secname}
  \textbf{Scenario.} Fictional HK bilingual kids' centre (\ctextbf{BrightNest / 亮巢學習中心}).
  \begin{itemize}
    \item Parents ask repetitive questions after hours: fees, hours, holidays, trials
    \item Staff cannot sit on WhatsApp 24/7; wrong fee/hours answers are costly
    \item Opportunity: an LLM assistant that is \cemph{grounded} in centre knowledge and can capture provisional bookings
  \end{itemize}
  \vspace{0.4em}
  \begin{block}{Training-free requirement (assignment)}
    We use \btextbf{RAG} (not fine-tuning) plus light structured tools --- docs can change without retraining.
  \end{block}
\end{frame}

\begin{frame}{Technique choice: RAG + tools}{\thesection \, \secname}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \textbf{Why RAG}
      \begin{itemize}
        \item KB changes weekly; no model retrain
        \item Auditable retrieve $\rightarrow$ generate path
        \item Citations in the UI (``Sources used'')
      \end{itemize}
      \vspace{0.5em}
      \textbf{Why also tools}
      \begin{itemize}
        \item Opening hours / HKO weather / public holidays
        \item Availability + booking / cancel dialogues
        \item Facts that should not be ``guessed'' by the LLM
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Self-built pipeline}
        \scriptsize
        chunk markdown\\
        $\downarrow$\\
        embed (DashScope)\\
        $\downarrow$\\
        Chroma top-$k$\\
        $\downarrow$\\
        Qwen generate\\[0.4em]
        \textit{Not a managed black-box KB.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Beat 2 --- RAG with visible grounding}{\thesection \, \secname}

\begin{frame}{Beat 3 --- Trial booking as a structured tool}{\thesection \, \secname}
  \textbf{Goal.} Capture a provisional after-hours booking without inventing seats.
  \begin{itemize}
    \item Identity from demo WhatsApp number $\rightarrow$ Demo Parent / Jason Chan
    \item Probe: \btexttt{Book Jason} \textit{or} Booking tool $\rightarrow$ Start booking dialogue
    \item Availability comes from CSV admin sheets (sessions / trial slots)
    \item Optional: \btexttt{Cancel class} for the multi-turn cancel path
  \end{itemize}
  \vspace{0.3em}
  \democue{%
    Switch to the demo $\rightarrow$ run booking for Jason.\\
    Show confirmation / booking id; note ``no payment --- hand-off for staff''.
  }
\end{frame}

\begin{frame}{Beat 4 --- Soft gaps become staff work}{\thesection \, \secname}
  \textbf{Goal.} Honest behaviour when the KB cannot ground an answer.
  \begin{itemize}
    \item Probe: \btexttt{Unknown ask} (or a niche question with weak coverage)
    \item Parent still gets a short, careful reply
    \item System appends a row to \texttt{admin\_todos.csv} for staff review
  \end{itemize}
  \vspace{0.4em}
  \democue{%
    Switch to the demo $\rightarrow$ \textbf{Unknown ask}.\\
    Expand \textbf{Admin to-do} and show the new open item.
  }
\end{frame}
