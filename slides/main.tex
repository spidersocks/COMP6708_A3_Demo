\begin{frame}{Technique: training-free RAG + tools}{\thesection}
  \textbf{Why training-free?} Centre docs change weekly $\Rightarrow$ fine-tuning is the wrong loop.
  \vspace{0.3em}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{RAG (self-built)}
      \begin{enumerate}\setlength\itemsep{0.18em}
        \item Chunk markdown KB (\texttt{content/})
        \item Embed with DashScope
        \item Store / query in Chroma
        \item Channel pack (3 dense + 2 BM25) $\rightarrow$ top-$k$
        \item Generate with Qwen + citations
      \end{enumerate}
      \vspace{0.3em}
      {\scriptsize Retrieval $s(q,c)=\mathrm{sim}(E(q),E(c))$ · inspector shows rank, $s$, chunk.}
    \end{column}
    \begin{column}{0.46\textwidth}
      \textbf{Hybrid router}
      \begin{itemize}\setlength\itemsep{0.16em}
        \item Scores = \textbf{detector awards}, not RAG sim
        \item Soft FAQ: grounded compose (card $>$ KB)
        \item Hard paths \textbf{bypass retrieve}
        \item Route arbiter if low confidence
        \item \textbf{Final check} before book / escalate / refuse
      \end{itemize}
      \vspace{0.35em}
      \pill{card privileged}\quad\pill{KB may enrich}\\[0.25em]
      \pill{proceed / clarify / abort}
    \end{column}
  \end{columns}
\end{frame}
